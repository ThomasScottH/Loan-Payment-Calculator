function calculatePayment() { // 3 inputs let loanAmount = Number(document.getElementById("loanAmount").value); let interestRate = Number(document.getElementById("interestRate").value) / 100 / 12; let loanTerm = Number(document.getElementById("loanTerm").value) * 12; // Calculate monthly payment let payment = loanAmount * interestRate * Math.pow(1 + interestRate, loanTerm) / (Math.pow(1 + interestRate, loanTerm) - 1); // 1 output document.getElementById("output").textContent = "$" + payment.toFixed(2); }
